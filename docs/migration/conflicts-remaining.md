      return "%";
    default:
      return {};
  }

static void button_edit_unit_hint_refresh(bContext *C, Button *but, HandleButtonData *data)
{
  /* Unit completion (hint) is only done for buttons with a unit or with a property such as
   * percentage or pixel for e.g. For everything else, we reset the completion to an empty string.
   */
  const PropertySubType subtype = but->rnaprop ? RNA_property_subtype(but->rnaprop) : PROP_NONE;
  const std::optional<StringRef> subtype_hint = button_edit_unit_hint_get_from_prop_subtype(
      subtype);
  if (!button_is_unit(but) && !subtype_hint.has_value()) {
    data->text_edit_unit_hint.clear();
    return;
  }

  /* Set the completion text (hint) to the unit that is used by this value. */
  std::string name_short;
  const int unit_type = RNA_SUBTYPE_UNIT_VALUE(button_unit_type_get(but));

    /* If the string contains the unit already, don't add it as a hint. */

      data->text_edit_unit_hint.clear();
      return;

      data->text_edit_unit_hint.clear();
      return;

    const void *usys;
    int len;
    UnitSettings &unit_settings = CTX_data_scene(C)->unit;
    BKE_unit_system_get(unit_settings.system, unit_type, &usys, &len);
    const int unit_index = BKE_preffered_unit_of_type_or_base_get(unit_settings, unit_type);
    name_short = BKE_unit_display_name_short_get(usys, unit_index);

/**
 * Close every Agent Bubble window except \a skip_win.
 *
 * Called from #wm_window_close when the last main window is being
 * closed — bubble windows must go first so the "last window" logic
 * doesn't see them as remaining main windows. Each bubble window
 * has `win->parent != nullptr`, so #wm_window_close skips the
 * "last main window" branch for them (no infinite recursion).
 *
 * The outer while-loop + inner break pattern restarts iteration
 * after each close because #wm_window_close mutates the window
 * list.
 */
static void wm_close_agent_bubble_windows(bContext *C,
                                           wmWindowManager *wm,
                                           wmWindow *skip_win)
{
  if (wm == nullptr) {
    return;
  }

  /* Save the context window — wm_window_close sets CTX_wm_window to the
   * window being closed (for handler teardown), then wm_window_free nulls
   * it.  Callers such as wm_exit_schedule_delayed expect a valid context
   * window after this function returns. */
  wmWindow *ctx_win = CTX_wm_window(C);

  bool closed_one = true;
  while (closed_one) {
    closed_one = false;
    LISTBASE_FOREACH (wmWindow *, iter_win, &wm->windows) {
      if (iter_win == skip_win || !wm_window_contains_agent_bubble_space(iter_win)) {
        continue;
      }
      wm_window_close(C, wm, iter_win);
      closed_one = true;
      break;
    }
  }

  /* Restore so downstream code still has a valid context window.
   *
   * ctx_win may have been freed above (it can be a bubble window, or a child
   * of one closed recursively by wm_window_close), so only restore it when it
   * is still in the window list — never put a freed pointer in the context.
   * Fall back to skip_win, which this function never closes. */

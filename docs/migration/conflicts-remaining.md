 * after each close because #wm_window_close mutates the window
 * list.

                                           wmWindowManager *wm,
                                           wmWindow *skip_win)

  if (wm == nullptr) {
    return;
  }

  /* Save the context window — wm_window_close sets CTX_wm_window to the
   * window being closed (for handler teardown), then wm_window_free nulls
   * it.  Callers such as wm_exit_schedule_delayed expect a valid context
   * window after this function returns. */

  bool closed_one = true;

    closed_one = false;
